What can embedded code see of the course? (Spoiler: nothing - it runs in an origin of its own.)

    require "js"
    def peek
      yield
    rescue Exception => e
      "blocked (#{e.class}: #{e.message[0, 60]})"
    end
    {
      origin: JS.global[:origin].to_s,
      local_storage: peek { JS.global[:localStorage].getItem("chunky_done").to_s },
      indexed_db: peek { JS.global[:indexedDB].open("probe"); "open" },
      cookie: peek { JS.global[:document][:cookie].to_s }
    }
